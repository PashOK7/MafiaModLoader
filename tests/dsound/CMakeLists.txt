cmake_minimum_required(VERSION 3.20)
project(mml_sound_restart_regression LANGUAGES CXX)
add_executable(dsound_repro restart.cpp)
target_compile_features(dsound_repro PRIVATE cxx_std_17)
target_compile_definitions(dsound_repro PRIVATE UNICODE _UNICODE NOMINMAX DIRECTSOUND_VERSION=0x0800)
target_link_libraries(dsound_repro PRIVATE dsound dxguid winmm user32)
add_executable(loopback loopback.cpp)
target_compile_features(loopback PRIVATE cxx_std_17)
target_compile_definitions(loopback PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(loopback PRIVATE ole32 uuid)
